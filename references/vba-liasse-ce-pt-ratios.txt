'Macro permettant d'ajouter une ligne à la fin du tableau, en préservant les formules
Public Sub DupliquerLigne10(codeTableau)

    Dim onglet, adresseCellule, Formule As String
    Dim i, nbLignes, premiereColonne, derniereColonne, derniereLigne As Integer
        
    'Demande le nombre de lignes à ajouter à l'utilisateur
    nbLignes = InputBox("Nombre de lignes à insérer (entre 1 et 500) : ", "Saisir le nombre de lignes")
    
    'Vérification de la valeur saisie
    If nbLignes = "" Then
        Exit Sub
    End If
    If nbLignes < 1 Or nbLignes > 500 Then
        Select Case MsgBox("Merci de saisir un nombre entre 1 et 500.", 0 + 64, "Erreur de saisie")
        End Select
        Exit Sub
    End If
    
    On Error Resume Next
    
    'Recherche des informations du tableau
    Set tableau = Sheets("tableaux").ListObjects("ListeTableaux").DataBodyRange.Columns(1).Find(codeTableau, LookAt:=xlWhole)
    onglet = Trim(tableau.Offset(0, 3).Text)
    
    'Désactive la protection
    Worksheets(onglet).Unprotect Password:="Menz12011956MSG"
    
    Set premiereCellule = Sheets(onglet).Range(Trim(tableau.Offset(0, 4).Text))
    Set derniereCellule = Sheets(onglet).Range(Trim(tableau.Offset(0, 5).Text))
    premiereColonne = premiereCellule.Column
    derniereColonne = derniereCellule.Column
    derniereLigne = derniereCellule.Row

    'Cas particulier des tableaux avec une ligne fixe à la fin
    If codeTableau = "FR2B" Then
        ligne = derniereLigne
    Else
        ligne = derniereLigne + 1
    End If
        
    For i = 1 To nbLignes
    
        'Ajout d'une nouvelle ligne
        Rows(ligne).Insert Shift:=xlDown
        
        'Copie du contenu de la ligne d'au-dessus
        Rows("10:10").RowHeight = 12
        Rows(10).EntireRow.Copy Rows(ligne)
      
        Application.CutCopyMode = False
        
        'Nettoyage pour enlever le contenu copié (excepté les formules)
                If premiereColonne = derniereColonne Then
            Sheets(onglet).Cells(ligne, premiereColonne).Select
        Else
            Sheets(onglet).Range(Cells(ligne, premiereColonne).Address & ":" & Cells(ligne, derniereColonne).Address).Select
        End If
        Intersect(Selection, _
            Selection.SpecialCells(xlCellTypeConstants)).ClearContents
    
        ligne = ligne + 1
    Next
    Rows("10:10").RowHeight = 0
    'Réactive la protection
   ActiveSheet.Protect Password:="Menz12011956MSG", DrawingObjects:=True, Contents:=True, Scenarios:=True, AllowFiltering:=True
End Sub

Function HdNum() As String
    Dim fsObj   As Object
    Dim drv     As Object
    Set fsObj = CreateObject("Scripting.FileSystemObject")
    Set drv = fsObj.Drives("C")
    HdNum = Hex(drv.SerialNumber)

Sub Remplacer_Image_avancement()
Application.ScreenUpdating = False
Application.Calculation = xlCalculationManual
Worksheets("MENU PRINCIPAL").Unprotect "Menz12011956MSG"
    Dim Emplacement As Range
    Dim Img As Object
    Dim ShapeObj As Shape
'Ouvrir l'explorateur de fichier
    If Application.Dialogs(xlDialogInsertPicture).Show Then
    Set Emplacement = Range("LOGO1") 'Définit l'emplacement de l'image
    Set Img = ActiveSheet.DrawingObjects(ActiveSheet.Shapes.count)
'Boucle pour supprimer l'ancienne image
    For Each ShapeObj In ActiveSheet.Shapes
        If ShapeObj.Name = "Image_Avancement" Then ActiveSheet.Shapes("Image_Avancement").Delete
    Next ShapeObj
    With Img.ShapeRange
        .Name = "Image_Avancement"          'Renommer l'image pour pouvoir la supprimer
        .LockAspectRatio = msoFalse
        .Top = Emplacement.Top + 2    'Si pas de chiffre = l'image sera aux mêmes dimensions que la cellule
        .Left = Emplacement.Left + 15    'Si pas de chiffre = l'image sera aux mêmes dimensions que la cellule
        .Width = Emplacement.Width - 30   'Si pas de chiffre = l'image sera aux mêmes dimensions que la cellule
        .Height = Emplacement.Height + 155 'Si pas de chiffre = l'image sera aux mêmes dimensions que la cellule
        Worksheets("MENU PRINCIPAL").Range("A29") = "Cliquez ici pour suprimer le logo"
        Sheets("MENU PRINCIPAL").Shapes("SupprimerLogo").Visible = True
        
   
        
    End With
    'Affecter la macro à l'image
        Selection.OnAction = "Remplacer_Image_avancement"
        [LOGO1].Select
        
     Worksheets("MENU PRINCIPAL").Shapes("Image_Avancement").Copy
     Worksheets("COUVERTURE").Activate
     Worksheets("COUVERTURE").Unprotect "Menz12011956MSG"
     Worksheets("COUVERTURE").Shapes("Logo").Delete
    Worksheets("COUVERTURE").Paste Range("E4")
    ActiveSheet.Shapes.Range(Array("Image_Avancement")).Select
    Selection.ShapeRange.ScaleWidth 0.5419630156, msoFalse, msoScaleFromTopLeft
    Selection.ShapeRange.ScaleHeight 0.5619630156, msoFalse, msoScaleFromTopLeft
    Selection.ShapeRange.IncrementTop -13.75
    Selection.OnAction = ""
    Worksheets("COUVERTURE").Protect "Menz12011956MSG"
   Worksheets("MENU PRINCIPAL").Activate
    Worksheets("MENU PRINCIPAL").Range("C1").Select
    
    Else
        MsgBox _
            "L'image n'a pas été insérée", vbInformation, "! Oups ! Action interrompue"
            
    End If
Application.Calculation = xlCalculationAutomatic
Application.ScreenUpdating = True
Worksheets("MENU PRINCIPAL").Protect "Menz12011956MSG"
End Sub

' ==============================================================================
' PROCÉDURE PRINCIPALE : LANCE L'EXPORTATION
' ==============================================================================
Sub Exporter_Vers_Excel_Cliquer()
     Dim wbSource As Workbook
    Dim wbNouveau As Workbook
    Dim ws As Worksheet
    Dim arrExclusions As Variant
    Dim nomFeuille As Variant
    Dim estExclue As Boolean
    Dim compteur As Integer
    
    ' 1. Optimisation : On fige l'écran pour la rapidité
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.Calculation = xlCalculationManual
    
    Set wbSource = ThisWorkbook
    
    ' 2. Liste des feuilles à EXCLURE
    arrExclusions = Array("PLAN DES COMPTES", "ACCUEIL", "GRAND-LIVRE N", _
                          "BALANCE N", "BALANCE N-1", "CONTRÔLE D'EQUILIBRE", _
                          "Contrôle des reports")
    
    ' 3. Création du nouveau classeur
    Set wbNouveau = Workbooks.Add(xlWBATWorksheet)
    
    
    compteur = 0
    
    ' 4. Boucle sur toutes les feuilles du classeur source
    For Each ws In wbSource.Worksheets
        ' Vérifie si visible
        If ws.Visible = xlSheetVisible Then
            estExclue = False
            
            ' Vérifie si exclue par son nom
            For Each nomFeuille In arrExclusions
                If UCase(ws.Name) = UCase(nomFeuille) Then
                    estExclue = True
                    Exit For
                End If
            Next nomFeuille
            
            ' Si OK, on traite
            If Not estExclue Then
                ' Copie de la feuille
                ws.Copy After:=wbNouveau.Sheets(wbNouveau.Sheets.count)
                
                ' Nettoyage complet de la feuille copiée
                Call TraiterEtNettoyerFeuille(wbNouveau.ActiveSheet)
                
                compteur = compteur + 1
            End If
        End If
    Next ws
    
    ' 5. Suppression de la feuille vide par défaut
    If wbNouveau.Sheets.count > 1 Then
        On Error Resume Next
        wbNouveau.Sheets(1).Delete
        On Error GoTo 0
    End If
    
    ' 6. Rétablissement des paramètres
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.Calculation = xlCalculationAutomatic
    
    
    ' 7. Recadrage de la zone d'imprission de la feuille SUPPL4
    With Sheets("SUPPL4").PageSetup
        ' Définit la plage de A à N
        .PrintArea = "$A:$N"
        
        ' Ajuste la largeur à une seule page
        .Zoom = False
        .FitToPagesWide = 1
        .FitToPagesTall = False ' Laisse la longueur s'étendre sur plusieurs pages si nécessaire
    End With
    
    
    ' 8. REACTIVATION DES FILTRES
    Sheets("NOTE36 Suite (Nomenclature)").Range("$o$11:$o$239").AutoFilter Field:=1, Criteria1:="1"
    Sheets("COMPTE D'EXPLOITATION REEL").Range("$N$6:$AH$601").AutoFilter Field:=2, Criteria1:="1"
    Sheets("COMPTE D'EXPLOITATION PREVISION").Range("$N$6:$AL$601").AutoFilter Field:=2, Criteria1:="1"
    Sheets("PLAN DE TRESORERIE REEL").Range("$N$5:$AH$600").AutoFilter Field:=2, Criteria1:="1"
    Sheets("PLAN DE TRESORERIE PREVISIONNEL").Range("$N$5:$AL$600").AutoFilter Field:=2, Criteria1:="1"
    
    ' 9. RETOUR A LA FEUILLE COUVERTURE
    Sheets("COUVERTURE").Activate
    
    ' 10. Message de fin
    If compteur > 0 Then
        MsgBox "Export terminé avec succès !" & vbCrLf & compteur & " feuilles exportées.", vbInformation, "Export Réussi"
    Else
        MsgBox "Aucune feuille correspondant aux critères n'a été trouvée.", vbExclamation, "Attention"
        wbNouveau.Close SaveChanges:=False
    End If
 End Sub

' ==============================================================================
' SOUS-PROCÉDURE : ORCHESTRE LE NETTOYAGE (CONSERVATION DU MODE FILTRE)
' ==============================================================================
Private Sub TraiterEtNettoyerFeuille(ws As Worksheet)
    Dim rngUsed As Range
    
    ' 1. Déverrouiller la feuille
    On Error Resume Next
    ws.Unprotect Password:=MOT_DE_PASSE
    On Error GoTo 0
    
    ' 2. GESTION DES FILTRES (RÉÉCRITURE : Affiche tout sans enlever les flèches)
    ' Si un filtre est appliqué (lignes masquées), on efface les critères
    If ws.FilterMode Then
        ws.ShowAllData
    End If
    
    ' Si l'AutoFilter n'était pas activé du tout, on pourrait l'activer sur la plage utilisée
    ' Mais si vous voulez garder la structure d'origine, on ne touche à rien d'autre.
    ' Les lignes masquées manuellement sont aussi réaffichées
    ws.Rows.Hidden = False
    
    ' 3. CONVERSION EN VALEURS
    ' Note : Le copier/coller en valeurs ne casse pas le mode filtre existant
    Set rngUsed = ws.UsedRange
    If Not rngUsed Is Nothing Then
        rngUsed.Copy
        rngUsed.PasteSpecial Paste:=xlPasteValues, Operation:=xlNone, SkipBlanks:=False, Transpose:=False
        Application.CutCopyMode = False
    End If
    
    ' 4. NETTOYAGE DES OBJETS / IMAGES (Sauf "Logo" et "Image")
    Call NettoyerObjets(ws)
    
    ' 5. RECADRAGE STRICT SUR LA ZONE D'IMPRESSION (Colonnes G/D)
    Call RecadrerSurZoneImpression(ws)
    
    ' 6. Remise à zéro de la position
    On Error Resume Next
    Application.GoTo ws.Range("A1"), True
    On Error GoTo 0
    
End Sub
' ==============================================================================
' SOUS-PROCÉDURE : SUPPRESSION DES OBJETS (IMAGES, FORMES...)
' ==============================================================================
Private Sub NettoyerObjets(ws As Worksheet)
    Dim Shp As Shape
    Dim i As Long
    
    ' On boucle à l'envers pour supprimer sans erreur d'index
    For i = ws.Shapes.count To 1 Step -1
        Set Shp = ws.Shapes(i)
        
        ' CONDITION 1 : L'objet doit être visible
        ' CONDITION 2 : Le nom ne doit être ni "LOGO" ni "IMAGE"
        If Shp.Visible = msoTrue Then
            If UCase(Shp.Name) <> "LOGO" And UCase(Shp.Name) <> "IMAGE" Then
                Shp.Delete
            End If
        End If
    Next i
End Sub
' ==============================================================================
' SOUS-PROCÉDURE : RECADRAGE COLONNES (GAUCHE ET DROITE)
' ==============================================================================
Private Sub RecadrerSurZoneImpression(ws As Worksheet)
    Dim strPrintArea As String
    Dim rngPrint As Range
    Dim area As Range
    Dim minCol As Long, maxCol As Long
    Dim minRow As Long
    Dim lastCol As Long
    
    strPrintArea = ws.PageSetup.PrintArea
    
    ' Si pas de zone d'impression, on quitte
    If strPrintArea = "" Then Exit Sub
    
    On Error Resume Next
    Set rngPrint = ws.Range(strPrintArea)
    On Error GoTo 0
    
    If rngPrint Is Nothing Then Exit Sub
    
    ' Initialisation pour trouver les limites extrêmes
    minCol = ws.Columns.count
    maxCol = 0
    minRow = ws.Rows.count
    
    ' Analyse de la zone d'impression (gère les zones multiples)
    For Each area In rngPrint.Areas
        If area.Column < minCol Then minCol = area.Column
        If (area.Column + area.Columns.count - 1) > maxCol Then maxCol = area.Column + area.Columns.count - 1
        If area.Row < minRow Then minRow = area.Row
    Next area
    
    ' --- SUPPRESSION DROITE ---
    lastCol = ws.UsedRange.Columns.count + ws.UsedRange.Column - 1
    If lastCol > maxCol Then
        ws.Range(ws.Columns(maxCol + 1), ws.Columns(ws.Columns.count)).Delete
    End If
    
    ' --- SUPPRESSION GAUCHE ---
    If minCol > 1 Then
        ws.Range(ws.Columns(1), ws.Columns(minCol - 1)).Delete
    End If
    
    ' --- SUPPRESSION HAUT (Entêtes hors zone) ---
    If minRow > 1 Then
        ws.Rows("1:" & (minRow - 1)).Delete
    End If
    
    ' On retire la définition de la zone d'impression pour finir propre
    ws.PageSetup.PrintArea = ""
End Sub

        SaveWithDocument:=msoTrue, _
        Left:=Emplacement.MergeArea.Left, _
        Top:=Emplacement.MergeArea.Top, _
        Width:=-1, Height:=-1) ' Taille réelle temporaire pour récupérer le ratio
        
    ' Paramétrage ACCUEIL avec conservation des proportions
    With nouvImg
        .Name = "Logo"
        .LockAspectRatio = msoTrue ' IMPORTANT : On garde les proportions
        
        ' --- ALGORITHME POUR FAIRE ENTRER L'IMAGE DANS S19 ---
        ' On essaie d'abord d'ajuster la largeur à celle de la cellule
        .Width = Emplacement.MergeArea.Width
        
        ' Si en ajustant la largeur, la hauteur dépasse la cellule, alors on ajuste par la hauteur
        If .Height > Emplacement.MergeArea.Height Then
            .Height = Emplacement.MergeArea.Height
        End If
        
        ' --- CENTRAGE DE L'IMAGE DANS S19 ---
        ' Calcul pour centrer horizontalement
        .Left = Emplacement.MergeArea.Left + ((Emplacement.MergeArea.Width - .Width) / 2)
        ' Calcul pour centrer verticalement
        .Top = Emplacement.MergeArea.Top + ((Emplacement.MergeArea.Height - .Height) / 2)
        
        .Placement = xlMoveAndSize
        .OnAction = "Remplacer_Logo"
    End With
    
    ' Gestion du texte et du bouton de suppression
    wsAcc.Range("S44").value = "Cliquez ici pour supprimer le logo"
    On Error Resume Next
    wsAcc.Shapes("SuppressionLogo").Visible = True
    On Error GoTo ErrorHandler
    
    ' --- PARTIE 3 : Insertion sur COUVERTURE (E4) ---
    wsCouv.Activate
    Set Emplacement = wsCouv.Range("E4")
    
    ' Supprimer l'ancien logo sur COUVERTURE s'il existe
    On Error Resume Next
    wsCouv.Shapes("Logo").Delete
    On Error GoTo ErrorHandler
    
    ' Insertion image
    Set nouvImg = wsCouv.Shapes.AddPicture( _
        FileName:=CheminImage, _
        LinkToFile:=msoFalse, _
        SaveWithDocument:=msoTrue, _
        Left:=Emplacement.Left, _
        Top:=Emplacement.Top, _
        Width:=100, Height:=100) ' Taille fixe temporaire
        
    ' Paramétrage COUVERTURE
    With nouvImg
        .Name = "Logo"
        .OnAction = ""
        
        ' 1. On remet l'image à sa taille réelle pour partir d'une base saine
        .ScaleWidth 1, msoTrue
        .ScaleHeight 1, msoTrue
        
        ' 2. On verrouille le ratio
        .LockAspectRatio = msoTrue ' IMPORTANT : On garde les proportions
        
        ' 3. On impose la largeur de 3,4 cm
        ' La hauteur va s'adapter toute seule grâce au LockAspectRatio
        .Width = Application.CentimetersToPoints(3.4)
        
        ' 4. Ajustement vertical (votre demande initiale)
        .IncrementTop -13.75
        
        ' 5. Ajustement horizontal (AJOUTÉ : 1 mm vers la droite)
        .IncrementLeft Application.CentimetersToPoints(0.1)
    End With
    
    ' Retour sur ACCUEIL et nettoyage
    wsAcc.Activate
    wsAcc.Range("L6").Select
    wsAcc.Range("S19").MergeArea.ClearContents

Sortie:
    ' Rétablissement de la protection
    wsCouv.Protect mdp
    wsAcc.Protect mdp
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
    Exit Sub

ErrorHandler:
    MsgBox "Une erreur est survenue : " & Err.Description, vbCritical
    Resume Sortie
End Sub

Sub ImorterGrandLivre_Cliquer()
ActiveSheet.Unprotect Password:="Menz12011956MSG"
    Dim cheminFichier As Variant
    Dim requete As WorkbookQuery
    Dim nouvelleSource As String
    Dim reponse As VbMsgBoxResult
    
    ' 1. Sélection du fichier
    cheminFichier = Application.GetOpenFilename( _
        FileFilter:="Fichiers Texte/CSV (*.txt;*.csv), *.txt;*.csv", _
        Title:="Choisir le fichier Grand Livre")
    
    If cheminFichier = False Then Exit Sub
    
    reponse = MsgBox("Importer depuis :" & vbCrLf & cheminFichier, vbOKCancel + vbQuestion)
    If reponse = vbCancel Then Exit Sub
    
    On Error Resume Next
    Set requete = ThisWorkbook.Queries("GrandLivreN")
    On Error GoTo 0
    
    If requete Is Nothing Then
        MsgBox "Requête 'GrandLivreN' introuvable.", vbCritical
        Exit Sub
    End If
    
    ' --- CONSTRUCTION DU CODE POWER QUERY ---
    nouvelleSource = ""
    nouvelleSource = nouvelleSource & "let" & vbCrLf
    nouvelleSource = nouvelleSource & "    Source = Csv.Document(File.Contents(""" & cheminFichier & """),[Delimiter=""#(tab)"", Columns=9, Encoding=1252, QuoteStyle=QuoteStyle.None])," & vbCrLf
    
    ' 1. On garde la Column2 en TEXTE pour l'instant (pour ne pas perdre le 0 du début : 010126)
    nouvelleSource = nouvelleSource & "    #""Type modifié"" = Table.TransformColumnTypes(Source,{" & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column1"", Int64.Type}, " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column2"", type text}, " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column3"", type text}, " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column4"", Int64.Type}, " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column5"", type text}, " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column6"", type text}, " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column7"", type number}, " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column8"", type number}, " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column9"", Int64.Type}}, ""fr-FR"")," & vbCrLf
    
    ' --- AJOUT DU TRAITEMENT DE LA DATE ICI ---
    ' On transforme "010126" en vraie Date. Format=""ddMMyy"" indique JourMoisAnnée sur 2 chiffres.
    nouvelleSource = nouvelleSource & "    #""TransformationDate"" = Table.TransformColumns(#""Type modifié"",{{""Column2"", each Date.FromText(_, [Format=""ddMMyy""]), type date}})," & vbCrLf
    
    ' Note : J'ai changé la source de l'étape suivante pour utiliser #""TransformationDate"" au lieu de #""Type modifié""
    nouvelleSource = nouvelleSource & "    #""Colonnes supprimées"" = Table.RemoveColumns(#""TransformationDate"", {""Column3"", ""Column4"", ""Column6"", ""Column9""})," & vbCrLf
    
    ' Renommage
    nouvelleSource = nouvelleSource & "    #""Colonnes renommées"" = Table.RenameColumns(#""Colonnes supprimées"", { " & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column1"", ""N° CPTE""}," & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column2"", ""DATE""}," & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column5"", ""LIBELLES""}," & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column7"", ""Md_GL""}," & vbCrLf
    nouvelleSource = nouvelleSource & "        {""Column8"", ""Mc_GL""}" & vbCrLf
    nouvelleSource = nouvelleSource & "    })" & vbCrLf
    
    nouvelleSource = nouvelleSource & "in" & vbCrLf
    nouvelleSource = nouvelleSource & "    #""Colonnes renommées"""
    
    ' Mise à jour et rafraîchissement
    requete.Formula = nouvelleSource
   
    Application.DisplayAlerts = False
    ThisWorkbook.Queries("GrandLivreN").Refresh
    On Error Resume Next
    ThisWorkbook.Queries("TraitementGrandLivre").Refresh
    On Error GoTo 0
    Application.DisplayAlerts = True
    
    MsgBox "Importation terminée. La colonne DATE est maintenant au format Date.", vbInformation

'Module d'aide à la préparation du mapping pour l'export de l'EDI XML
'Macros permettant d'obtenir les cellules correspondant à la position des champs et colonnes définie dans le mapping
'

Public Sub GenererMappingFixe()
    
    Dim codeTableau, codeChamp, positionLigne, positionColonne, onglet As String
    
    'Parcours des champs
    For Each celluleCodeTableau In Sheets("champs fixes").ListObjects("ListeChampsTableauxFixes").ListColumns(1).DataBodyRange
    
        codeTableau = Trim(celluleCodeTableau.Text)
        codeChamp = Trim(celluleCodeTableau.Offset(0, 4).Text)
        positionLigne = Trim(celluleCodeTableau.Offset(0, 5).Text)
        positionColonne = Trim(celluleCodeTableau.Offset(0, 6).Text)
        
        'Recherche des informations du tableau
        Set tableau = Sheets("tableaux").ListObjects("ListeTableaux").DataBodyRange.Columns(1).Find(codeTableau, LookAt:=xlWhole)
        onglet = Trim(tableau.Offset(0, 3).Text)
        Set premiereCellule = Sheets(onglet).Range(Trim(tableau.Offset(0, 4).Text))
        
        'Remplissage du mapping
        Set cellule = Sheets(onglet).Cells(premiereCellule.Row + positionLigne - 1, premiereCellule.Column + positionColonne - 1)
        celluleCodeTableau.Offset(0, 7) = cellule.Address(RowAbsolute:=False, ColumnAbsolute:=False) 'Cellule
        celluleCodeTableau.Offset(0, 8) = Trim(cellule.Text) 'Valeur

    Next celluleCodeTableau

End Sub

Public Sub GenererMappingVariable()

    Dim codeTableau, codeColonne, positionColonne, onglet As String
    
    'Parcours des champs
    For Each celluleCodeTableau In Sheets("champs variables").ListObjects("ListeColonnesTableauxVariables").ListColumns(1).DataBodyRange
    
        codeTableau = Trim(celluleCodeTableau.Text)
        codeColonne = Trim(celluleCodeTableau.Offset(0, 2).Text)
        positionColonne = Trim(celluleCodeTableau.Offset(0, 3).Text)
        
        'Recherche des informations du tableau
        Set tableau = Sheets("tableaux").ListObjects("ListeTableaux").DataBodyRange.Columns(1).Find(codeTableau, LookAt:=xlWhole)
        onglet = Trim(tableau.Offset(0, 3).Text)
        Set premiereCellule = Sheets(onglet).Range(Trim(tableau.Offset(0, 4).Text))
        
        'Remplissage du mapping
        Set cellule = Sheets(onglet).Cells(premiereCellule.Row, premiereCellule.Column + positionColonne - 1)
        celluleCodeTableau.Offset(0, 4) = cellule.Address(RowAbsolute:=False, ColumnAbsolute:=False) 'Cellule

    Next celluleCodeTableau
    
End Sub

===== RemplacerLogo.bas =====
Sub Remplacer_Logo()
Dim wsAcc As Worksheet, wsCouv As Worksheet
    Dim Emplacement As Range
    Dim nouvImg As Shape
    Dim mdp As String
    Dim Fd As FileDialog
    Dim CheminImage As String
    Dim RatioImage As Double, RatioCellule As Double
    
    ' Configuration
    mdp = "Menz12011956MSG"
    Set wsAcc = ThisWorkbook.Sheets("ACCUEIL")
    Set wsCouv = ThisWorkbook.Sheets("COUVERTURE")
    
    On Error GoTo ErrorHandler
    
    ' Optimisation de l'affichage
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    
    ' Déprotection des feuilles
    wsAcc.Unprotect mdp
    wsCouv.Unprotect mdp
    
    ' --- PARTIE 1 : Choix du fichier (Explorateur Windows) ---
    Set Fd = Application.FileDialog(msoFileDialogFilePicker)
    With Fd
        .Title = "Sélectionnez votre logo"
        .Filters.Clear
        .Filters.Add "Images", "*.jpg; *.jpeg; *.png; *.bmp; *.gif"
        .AllowMultiSelect = False
        
        If .Show = -1 Then
            CheminImage = .SelectedItems(1)
        Else
            MsgBox "Aucune image sélectionnée.", vbExclamation
            GoTo Sortie
        End If
    End With

    ' --- PARTIE 2 : Insertion sur ACCUEIL (S19 Fusionnée) ---
    wsAcc.Activate
    Set Emplacement = wsAcc.Range("S19")
    
    ' Supprimer l'ancien logo sur ACCUEIL s'il existe
    On Error Resume Next
    wsAcc.Shapes("Logo").Delete
    On Error GoTo ErrorHandler
    
    ' Insertion de l'image
    Set nouvImg = wsAcc.Shapes.AddPicture( _
        FileName:=CheminImage, _
        LinkToFile:=msoFalse, _
        SaveWithDocument:=msoTrue, _
        Left:=Emplacement.MergeArea.Left, _
        Top:=Emplacement.MergeArea.Top, _
        Width:=-1, Height:=-1) ' Taille réelle temporaire pour récupérer le ratio
        
    ' Paramétrage ACCUEIL avec conservation des proportions
    With nouvImg
        .Name = "Logo"
        .LockAspectRatio = msoTrue ' IMPORTANT : On garde les proportions
        
        ' --- ALGORITHME POUR FAIRE ENTRER L'IMAGE DANS S19 ---
        ' On essaie d'abord d'ajuster la largeur à celle de la cellule
        .Width = Emplacement.MergeArea.Width
        
        ' Si en ajustant la largeur, la hauteur dépasse la cellule, alors on ajuste par la hauteur
        If .Height > Emplacement.MergeArea.Height Then
            .Height = Emplacement.MergeArea.Height
        End If
        
        ' --- CENTRAGE DE L'IMAGE DANS S19 ---
        ' Calcul pour centrer horizontalement
        .Left = Emplacement.MergeArea.Left + ((Emplacement.MergeArea.Width - .Width) / 2)
        ' Calcul pour centrer verticalement
        .Top = Emplacement.MergeArea.Top + ((Emplacement.MergeArea.Height - .Height) / 2)
        
        .Placement = xlMoveAndSize
        .OnAction = "Remplacer_Logo"
    End With
    
    ' Gestion du texte et du bouton de suppression
    wsAcc.Range("S44").value = "Cliquez ici pour supprimer le logo"
    On Error Resume Next
    wsAcc.Shapes("SuppressionLogo").Visible = True
    On Error GoTo ErrorHandler
    
    ' --- PARTIE 3 : Insertion sur COUVERTURE (E4) ---
    wsCouv.Activate
    Set Emplacement = wsCouv.Range("E4")
    
    ' Supprimer l'ancien logo sur COUVERTURE s'il existe
    On Error Resume Next
    wsCouv.Shapes("Logo").Delete
    On Error GoTo ErrorHandler
    
    ' Insertion image
    Set nouvImg = wsCouv.Shapes.AddPicture( _
        FileName:=CheminImage, _
        LinkToFile:=msoFalse, _
        SaveWithDocument:=msoTrue, _
        Left:=Emplacement.Left, _
        Top:=Emplacement.Top, _
        Width:=100, Height:=100) ' Taille fixe temporaire
        
    ' Paramétrage COUVERTURE
    With nouvImg
        .Name = "Logo"
        .OnAction = ""
        
        ' 1. On remet l'image à sa taille réelle pour partir d'une base saine
        .ScaleWidth 1, msoTrue
        .ScaleHeight 1, msoTrue
        
        ' 2. On verrouille le ratio
        .LockAspectRatio = msoTrue ' IMPORTANT : On garde les proportions
        
        ' 3. On impose la largeur de 3,4 cm
        ' La hauteur va s'adapter toute seule grâce au LockAspectRatio
        .Width = Application.CentimetersToPoints(3.4)
        
        ' 4. Ajustement vertical (votre demande initiale)
        .IncrementTop -13.75
        
        ' 5. Ajustement horizontal (AJOUTÉ : 1 mm vers la droite)
        .IncrementLeft Application.CentimetersToPoints(0.1)
    End With
    
    ' Retour sur ACCUEIL et nettoyage
    wsAcc.Activate
    wsAcc.Range("L6").Select
    wsAcc.Range("S19").MergeArea.ClearContents

Sortie:
    ' Rétablissement de la protection
    wsCouv.Protect mdp
    wsAcc.Protect mdp
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
    Exit Sub

ErrorHandler:
    MsgBox "Une erreur est survenue : " & Err.Description, vbCritical
    Resume Sortie
End Sub

Sub Teleliasse()
    Dim Sh As Worksheet
    Dim Obj As Shape
    Dim i As Integer

    ' 1. Préparation et Déprotection
    ThisWorkbook.Save
    
    With ActiveWorkbook
        .Unprotect "Menz12011956MSG"
        
        ' 2. Boucle sur toutes les feuilles du classeur
        For Each Sh In .Worksheets
            
            ' Rendre la feuille visible par défaut (ou selon vos critères spécifiques)
            Sh.Visible = xlSheetVisible
            
            ' --- GESTION SPÉCIFIQUE DE LA VISIBILITÉ DES FEUILLES ---
            ' On utilise Select Case pour plus de clarté que de multiples lignes Sheets().Visible
            Select Case Sh.Name
                Case "CODES ACTIVITES", "R4", "Codes R2", "Plan Comptable", "TBFT", _
                     "export", "champs variables", "champs fixes", "tableaux", "Referentiels"
                    Sh.Visible = Range("N9").value

                     "COMPTE D'EXPLOITATION REEL", "COMPTE D'EXPLOITATION PREVISION", _

                    Sh.Visible = Range("EX1").value
                
                Case "Format Export", "Format Import"
                    Sh.Visible = Range("A57").value
                
                Case "ACCUEIL"
                    Sh.Visible = Range("AR23").value
            End Select

            ' On ne traite les objets que si la feuille est visible
            If Sh.Visible = xlSheetVisible Then
                For Each Obj In Sh.Shapes

                    If Obj.Visible = msoTrue Then

'Module de génération de fichiers EDI XML pour les états financiers
'La macro GenererXML permet de créer un fichier XML respectant la norme EDI e-impots à partir des données saisies dans les tableaux du classeur EXCEL
'

Public Sub GenererXML()
    Dim ChoixDossier As FileDialog ' Ajout de la variable pour la fenêtre de choix
    Dim Repertoire As String
    Dim NomFichier As String
    Dim mapToExport As XmlMap
    Dim ncc As Range, exercice As Range

    Application.Calculation = xlCalculationManual
    Application.ScreenUpdating = False
    
    Call Déverrouillage.Déprot
    Sheets("Export XML").Activate
    
    'Vérifie qu'un NCC et un exercice ont été renseignés
    Set ncc = Sheets("Export XML").Range("G11")
    Set exercice = Sheets("Export XML").Range("G12")
    
    If Trim(ncc.Text & vbNullString) = vbNullString Then
        MsgBox "Merci de remplir le champ NCC.", 0 + 64, "Impossible de générer le XML"
        GoTo Nettoyage ' Utilise un point de sortie pour rétablir les paramètres Excel
    End If
    
    If Trim(exercice.Text & vbNullString) = vbNullString Then
        MsgBox "Merci de remplir le champ exercice.", 0 + 64, "Impossible de générer le XML"
        GoTo Nettoyage
    End If
    
    'Génère la liste des champs dans l'onglet "export"
    Call GenererListeChamps
    
    ' --- MODIFICATION : CHOIX DU RÉPERTOIRE PAR L'UTILISATEUR ---
    Set ChoixDossier = Application.FileDialog(msoFileDialogFolderPicker)
    ChoixDossier.Title = "Sélectionnez le dossier où enregistrer le fichier XML"
    
    If ChoixDossier.Show = -1 Then
        Repertoire = ChoixDossier.SelectedItems(1)
    Else
        MsgBox "Génération annulée : aucun dossier sélectionné.", vbExclamation
        GoTo Nettoyage
    End If
    ' ----------------------------------------------------------
    
    'Définition du nom du fichier
    NomFichier = "NO-" & Replace(exercice, " ", "") & "-" & Replace(ncc, " ", "") & "-" & Format(Now, "ddmmyyyy-hhnnss") & ".xml"
    
    'Export à l'aide du fichier XSD lié à ce fichier Excel
    Set mapToExport = ActiveWorkbook.XmlMaps("EDI_Mappage")
    
    ' Sauvegarde avec gestion d'erreur au cas où le dossier serait protégé en écriture
    On Error Resume Next
    ActiveWorkbook.SaveAsXMLData Repertoire & Application.PathSeparator & NomFichier, mapToExport
    
    If Err.Number <> 0 Then
        MsgBox "Erreur lors de l'enregistrement : " & Err.Description, vbCritical, "Erreur"
        Err.Clear
    Else
        MsgBox "Votre fichier EDI a bien été généré." & vbNewLine & vbNewLine & _
               "Il a été enregistré dans le répertoire suivant :" & vbNewLine & Repertoire & _
               vbNewLine & vbNewLine & "Nom du fichier : " & NomFichier, 0 + 64, "Confirmation"
    End If
    On Error GoTo 0

Nettoyage:
    Call Verrouillage.Prot
    Sheets("Export XML").Select
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
End Sub

Public Sub GenererListeChampsVariables()
    Dim i, iLigne, nbLignes As Integer
    Dim codeTableau, codeColonne, positionColonne, onglet As String
    
    'Supprime toutes les lignes de la liste
    Call ResetTableau("ExportChampsTableauxVariables")
    
    i = 1
    
    'Parcours des colonnes
    For Each celluleCodeTableau In Sheets("champs variables").ListObjects("ListeColonnesTableauxVariables").ListColumns(1).DataBodyRange
    
        codeTableau = Trim(celluleCodeTableau.Text)
        codeColonne = Trim(celluleCodeTableau.Offset(0, 2).Text)
        positionColonne = Trim(celluleCodeTableau.Offset(0, 3).Text)
        
        'Recherche des informations du tableau
        Set tableau = Sheets("tableaux").ListObjects("ListeTableaux").DataBodyRange.Columns(1).Find(codeTableau, LookAt:=xlWhole)
        onglet = Trim(tableau.Offset(0, 3).Text)
        Set premiereCellule = Sheets(onglet).Range(Trim(tableau.Offset(0, 4).Text))
        Set derniereCellule = Sheets(onglet).Range(Trim(tableau.Offset(0, 5).Text))
        nbLignes = derniereCellule.Row - premiereCellule.Row + 1
        
        Set celluleCodeColonne = Sheets("export").Range("A15")
        
        'Parcours des lignes
        For iLigne = 1 To nbLignes
            
            Set cellule = Sheets(onglet).Cells(premiereCellule.Row + iLigne - 1, premiereCellule.Column + positionColonne - 1)
            
            If Not Trim(cellule.Text & vbNullString) = vbNullString Then 'Si le champ a été rempli
            
                'Ajout du champ à la liste
                celluleCodeColonne.Offset(i - 1, 0).value = codeColonne 'Colonne
                celluleCodeColonne.Offset(i - 1, 1).value = iLigne 'Ligne
                celluleCodeColonne.Offset(i - 1, 2).value = GetValeur(cellule.value) 'Valeur
                
                i = i + 1
                
            End If
        Next
        

    Next celluleCodeTableau
    
    'Mise à jour de la taille de la liste
    Set listeChamps = Sheets("export").ListObjects("ExportChampsTableauxVariables")
    listeChamps.Resize Range("A14:C" & CStr(14 + i - 1))
    
End Sub

Public Sub ResetTableau(nomTableau)

    Dim tableau As ListObject

    Set tableau = Sheets("export").ListObjects(nomTableau)
    
    'Suppression du contenu du tableau
    If Not tableau.DataBodyRange Is Nothing Then

    End If

Function GetValeur(valeur As String)
    If IsDate(valeur) And Not IsNumeric(valeur) Then
        GetValeur = "'" & Format(valeur, "dd/mm/yyyy")
    ElseIf IsNumeric(valeur) Then
        If valeur Like "0#*" Then 'Un zéro avec au moins un autre chiffre derrière : c'est une chaîne de caractères
            GetValeur = "'" & Trim(valeur)
        Else 'C'est un nombre : on arrondit à 4 décimales pour les pourcentages
            GetValeur = Round(valeur, 4)
        End If
    Else
        GetValeur = "'" & Trim(valeur)
    End If
End Function

'Macro permettant d'ajouter une ligne à la fin du tableau, en préservant les formules
Public Sub AjouterLigne(codeTableau)

    Dim onglet, adresseCellule, Formule As String
    Dim i, nbLignes, premiereColonne, derniereColonne, derniereLigne As Integer
        
    'Demande le nombre de lignes à ajouter à l'utilisateur
    nbLignes = InputBox("Nombre de lignes à insérer (entre 1 et 500) : ", "Saisir le nombre de lignes")
    
    'Vérification de la valeur saisie
    If nbLignes = "" Then
        Exit Sub
    End If
    If nbLignes < 1 Or nbLignes > 500 Then
        Select Case MsgBox("Merci de saisir un nombre entre 1 et 500.", 0 + 64, "Erreur de saisie")
        End Select
        Exit Sub
    End If
    
    On Error Resume Next
    
    'Recherche des informations du tableau
    Set tableau = Sheets("tableaux").ListObjects("ListeTableaux").DataBodyRange.Columns(1).Find(codeTableau, LookAt:=xlWhole)
    onglet = Trim(tableau.Offset(0, 3).Text)
    
    'Désactive la protection
    Worksheets(onglet).Unprotect Password:="Menz12011956MSG"
    
    Set premiereCellule = Sheets(onglet).Range(Trim(tableau.Offset(0, 4).Text))
    Set derniereCellule = Sheets(onglet).Range(Trim(tableau.Offset(0, 5).Text))
    premiereColonne = premiereCellule.Column
    derniereColonne = derniereCellule.Column
    derniereLigne = derniereCellule.Row

    'Cas particulier des tableaux avec une ligne fixe à la fin
    If codeTableau = "NOTE13B" Or codeTableau = "FR2B" Or codeTableau = "NOTE32" Or codeTableau = "NOTE33" Or codeTableau = "SUPPL1" Then
        ligne = derniereLigne
    Else
        ligne = derniereLigne + 1
    End If
        
    For i = 1 To nbLignes
    
        'Ajout d'une nouvelle ligne
        Rows(ligne).Insert Shift:=xlDown
        
        'Copie du contenu de la ligne d'au-dessus
        Rows(ligne - 1).EntireRow.Copy Rows(ligne)
        Application.CutCopyMode = False
        
        'Nettoyage pour enlever le contenu copié (excepté les formules)
        Sheets(onglet).Range(Cells(ligne, premiereColonne).Address & ":" & Cells(ligne, derniereColonne).Address).SpecialCells(xlCellTypeConstants).ClearContents
    
        ligne = ligne + 1
    Next
    
    'Mise à jour de la valeur de derniere cellule
    adresseCellule = Cells(derniereLigne + nbLignes, derniereColonne).Address(RowAbsolute:=False, ColumnAbsolute:=False)
    Formule = "=ADDRESS(ROW('" & onglet & "'!" & adresseCellule & "),COLUMN('" & onglet & "'!" & adresseCellule & "),4)"
    tableau.Offset(0, 5).Formula = Formule
    
    'Réactive la protection
     ActiveSheet.Protect Password:="Menz12011956MSG", DrawingObjects:=True, Contents:=True, Scenarios:=True, AllowFiltering:=True

End Sub
